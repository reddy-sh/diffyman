def read_rows(path):
  with open(path, encoding="utf-8") as handle:
    return parse_rows(handle)


def count_rows(path):
  return len(read_rows(path))


def parse_rows(lines):
  rows = []
  for raw_line in lines:
    line = raw_line.strip()
    if line and not line.startswith("#"):
      rows.append(line.split(","))
  return rows
